     ```

       if (isRealAdmin) {
           java.util.List<ChatSession> list = historyService.getAllSessionsForAdmin(limit);
           return list.stream()
                   .map(s -> new SessionInfo(
                           s.getId(),
                           s.getTitle(),
                           ChatModelMetaSupport.safeTrim(s.getLastAnswerMode()),
                           s.getLastTraceTurnId()))
                   .toList();
       }
       ```
